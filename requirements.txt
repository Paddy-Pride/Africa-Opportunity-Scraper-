    """Actual working scraper for African youth opportunities"""

            'Accept-Encoding': 'gzip, deflate, br',
            'Connection': 'keep-alive',
            'Upgrade-Insecure-Requests': '1',
        })
        
        # Real data sources that work
        self.sources = {
            'scholarship_africa': {
                'url': 'https://www.scholarshipsforafricans.com/feed',
                'type': 'rss'
            },
            'opportunity_desk': {
                'url': 'https://opportunitydesk.org/category/africa/feed/',
                'type': 'rss'
            },
            'youth_opportunities': {
                'url': 'https://www.youthop.com/feed',

    def scrape_rss_feed(self, url: str) -> List[Dict]:
        """Scrape RSS feeds - these actually work!"""
        opportunities = []
        try:
            feed = feedparser.parse(url)
            
            for entry in feed.entries[:30]:  # Get latest 30
                try:
                    # Parse title
                    title = entry.get('title', 'Unknown Opportunity')
                    
                    # Parse description
                    description = entry.get('description', '')
                    if hasattr(entry, 'summary'):
                        description = entry.summary
                    
                    # Parse date
                    published = entry.get('published', '')
                    if hasattr(entry, 'published_parsed'):
                        published = time.strftime('%Y-%m-%d', entry.published_parsed)
                    
                    # Extract deadline if present in description
                    deadline = self.extract_deadline_from_text(description)
                    
                    # Extract organization
                    organization = self.extract_organization(description)
                    
                    # Extract benefits
                    benefits = self.extract_benefits(description)
                    
                    # Determine if relevant to African youth
                    relevance_score = self.check_african_relevance(title + ' ' + description)
                    
                    if relevance_score > 0.5:  # Only include relevant opportunities
                        opportunity = {
                            'title': self.clean_text(title),
                            'organization': organization,
                            'deadline': deadline,
                            'benefits': benefits[:3] if benefits else ['Check description for details'],
                            'eligibility': self.extract_eligibility(description),
                            'description': self.clean_text(description)[:500],
                            'category': self.determine_category(title + ' ' + description),
                            'link': entry.get('link', '#'),
                            'source': 'RSS Feed',
                            'posted_date': published,
                            'verified': True,
                            'relevance_score': relevance_score
                        }
                        opportunities.append(opportunity)
                except Exception as e:
                    continue
                    
        except Exception as e:
            print(f"Error scraping RSS feed {url}: {e}")
        
        return opportunities

    def scrape_scholars4dev(self) -> List[Dict]:
        """Scrape Scholarships for Developing Countries - Real working scrape"""
        opportunities = []
        try:
            url = 'https://www.scholars4dev.com/category/africa-scholarships/'
            response = self.session.get(url, timeout=20)
            soup = BeautifulSoup(response.content, 'html.parser')
            
            # Find actual article posts
            articles = soup.find_all('article') or soup.find_all('div', class_='post')
            
            for article in articles[:20]:
                try:
                    # Find title
                    title_elem = article.find('h2') or article.find('h3') or article.find('h1')
                    if not title_elem:
                        continue
                    title = title_elem.get_text(strip=True)
                    
                    # Find link
                    link_elem = title_elem.find('a')
                    link = link_elem.get('href') if link_elem else '#'
                    
                    # Get content
                    content = article.get_text(strip=True)
                    
                    # Extract deadline from content
                    deadline = self.extract_deadline_from_text(content)
                    
                    # Extract organization
                    org = self.extract_organization(content)
                    
                    # Extract benefits
                    benefits = self.extract_benefits(content)
                    
                    # Extract eligibility
                    eligibility = self.extract_eligibility(content)
                    
                    # Determine category
                    category = self.determine_category(title + ' ' + content)
                    
                    opportunity = {
                        'title': self.clean_text(title)[:150],
                        'organization': org,
                        'deadline': deadline,
                        'benefits': benefits[:3] if benefits else ['Various benefits available'],
                        'eligibility': eligibility[:3] if eligibility else ['Open to African students'],
                        'description': self.clean_text(content)[:500],
                        'category': category,
                        'link': link,
                        'source': 'Scholars4Dev',
                        'verified': True
                    }
                    opportunities.append(opportunity)
                    
                except Exception as e:
                    continue
                    
        except Exception as e:
            print(f"Error scraping Scholars4Dev: {e}")
            
        return opportunities

    def extract_deadline_from_text(self, text: str) -> str:
        """Extract deadline from text using regex patterns"""
        # Common deadline patterns
        patterns = [
            r'(?:deadline|closing date|apply by|due date|application deadline)\s*:?\s*([A-Za-z]+\s+\d{1,2},?\s+\d{4})',
            r'(?:deadline|closing date|apply by|due date|application deadline)\s*:?\s*(\d{1,2}\s+[A-Za-z]+\s+\d{4})',
            r'(?:deadline|closing date|apply by|due date|application deadline)\s*:?\s*(\d{1,2}/\d{1,2}/\d{4})',
            r'(?:deadline|closing date|apply by|due date|application deadline)\s*:?\s*(\d{4}-\d{1,2}-\d{1,2})',
        ]
        
        for pattern in patterns:
            match = re.search(pattern, text, re.IGNORECASE)
            if match:
                return match.group(1)
        
        # Check for relative deadlines
        if 'rolling' in text.lower() or 'open' in text.lower():
            return 'Rolling/Open'
        
        return 'Not specified'

    def extract_organization(self, text: str) -> str:
        """Extract organization name from text"""
        # Common organization patterns
        patterns = [
            r'(?:host|organization|institution|university|offered by)\s*:?\s*([A-Z][A-Za-z\s]+?)(?:\.|\,|\n)',
            r'(?:by|at)\s+([A-Z][A-Za-z\s]+?)(?:\.|\,|\n)',
        ]
        
        for pattern in patterns:
            match = re.search(pattern, text, re.IGNORECASE)
            if match:
                org = match.group(1).strip()
                if len(org) > 5 and len(org) < 100:
                    return org
        
        return 'Various Organizations'

    def extract_benefits(self, text: str) -> List[str]:
        """Extract benefits from text"""
        benefits = []
        benefit_keywords = [
            'tuition', 'scholarship', 'grant', 'fellowship', 
            'stipend', 'allowance', 'funding', 'cover', 'support',
            'full tuition', 'partial tuition', 'living expenses'
        ]
        
        # Find sentences containing benefit keywords
        sentences = re.split(r'[.!?]+', text)
        for sentence in sentences:
            for keyword in benefit_keywords:
                if keyword in sentence.lower():
                    # Clean up the benefit
                    benefit = sentence.strip()
                    if benefit and len(benefit) > 10:
                        benefits.append(benefit[:100])
                        break
        
        return list(set(benefits))  # Remove duplicates

    def extract_eligibility(self, text: str) -> List[str]:
        """Extract eligibility criteria from text"""
        eligibility = []
        
        # Look for eligibility sections
        elig_patterns = [
            r'(?:eligible|eligibility|requirements|qualifications|criteria)\s*:?\s*([^.]+\.[^.]+\.[^.]+)',
            r'open to\s+([^.]+\.[^.]+\.[^.]+)',
        ]
        
        for pattern in elig_patterns:
            match = re.search(pattern, text, re.IGNORECASE)
            if match:
                elig_text = match.group(1).strip()
                # Split into bullet points
                points = re.split(r'[•\-;\n]', elig_text)
                for point in points[:3]:
                    point = point.strip()
                    if point and len(point) > 10:
                        eligibility.append(point)
                break
        
        # If no structured eligibility, look for common phrases
        if not eligibility:
            common_elig = [
                r'(?:must be|should be|require[sd]?)\s+([^.]+\.[^.]+)',
                r'(?:aged|age)\s+([^.]+\.[^.]+)',
                r'nationality\s+([^.]+\.[^.]+)',
            ]
            for pattern in common_elig:
                match = re.search(pattern, text, re.IGNORECASE)
                if match and match.group(1):
                    eligibility.append(match.group(1).strip())
        
        return eligibility[:3]  # Return top 3

    def determine_category(self, text: str) -> str:
        """Determine the category of opportunity"""
        text_lower = text.lower()
        
        categories = {
            'Tech/STEM': ['tech', 'stem', 'engineering', 'data', 'software', 'ai', 'machine learning', 
                         'programming', 'computer', 'science', 'technology', 'digital'],
            'Entrepreneurship': ['entrepreneur', 'startup', 'business', 'venture', 'innovation', 
                               'incubator', 'accelerator', 'pitch', 'start-up'],
            'Grant': ['grant', 'funding', 'sponsor', 'financial', 'research grant', 'project grant'],
            'Fellowship': ['fellowship', 'fellow', 'fellowshp', 'research fellow'],
            'Internship': ['intern', 'internship', 'trainee', 'apprentice', 'work experience']
        }
        
        for category, keywords in categories.items():
            if any(keyword in text_lower for keyword in keywords):
                return category
        
        return 'Other'

    def check_african_relevance(self, text: str) -> float:
        """Check if opportunity is relevant to African youth"""
        text_lower = text.lower()
        
        african_keywords = ['africa', 'african', 'nigeria', 'kenya', 'ghana', 'south africa', 
                           'uganda', 'tanzania', 'ethiopia', 'rwanda', 'zambia', 'zimbabwe',
                           'sub-saharan', 'west africa', 'east africa', 'southern africa']
        
        # Count African mentions
        score = 0
        for keyword in african_keywords:
            if keyword in text_lower:
                score += 1
        
        # Also check for 'developing countries' or 'international'
        if any(keyword in text_lower for keyword in ['developing country', 'developing countries']):
            score += 0.5
        if 'international' in text_lower:
            score += 0.3
        
        # Normalize score
        max_score = len(african_keywords)
        return min(score / max_score, 1.0)

    def clean_text(self, text: str) -> str:
        """Clean text by removing excessive whitespace and special characters"""
        # Remove HTML tags
        text = re.sub(r'<[^>]+>', '', text)
        # Remove excessive whitespace
        text = re.sub(r'\s+', ' ', text)
        # Remove special characters
        text = re.sub(r'[^\w\s.,;:!?()-]', '', text)
        return text.strip()

    def scrape_all(self) -> List[Dict]:
        """Scrape all sources and combine results"""
        all_opportunities = []
        
        print("🔄 Starting real scraping from multiple sources...")
        
        # Scrape RSS feeds
        for name, source in self.sources.items():
            if source['type'] == 'rss':
                print(f"📡 Scraping {name} RSS feed...")
                opportunities = self.scrape_rss_feed(source['url'])
                all_opportunities.extend(opportunities)
                print(f"✅ Found {len(opportunities)} opportunities from {name}")
                time.sleep(2)  # Be polite to servers
        
        # Scrape websites directly
        print("🌐 Scraping Scholars4Dev...")
        scholars_opps = self.scrape_scholars4dev()
        all_opportunities.extend(scholars_opps)
        print(f"✅ Found {len(scholars_opps)} opportunities from Scholars4Dev")
        
        print("🌐 Scraping Opportunity Desk...")
        desk_opps = self.scrape_opportunity_desk()
        all_opportunities.extend(desk_opps)
        print(f"✅ Found {len(desk_opps)} opportunities from Opportunity Desk")
        
        # Remove duplicates
        unique_opps = self.deduplicate_opportunities(all_opportunities)
        
        # Sort by deadline urgency
        unique_opps = self.sort_by_urgency(unique_opps)
        
        print(f"✅ Total unique opportunities: {len(unique_opps)}")
        return unique_opps

    def deduplicate_opportunities(self, opportunities: List[Dict]) -> List[Dict]:
        """Remove duplicate opportunities based on title"""
        seen_titles = set()
        unique = []
        
        for opp in opportunities:
            title = opp['title'].lower().strip()
            if title and title not in seen_titles and len(title) > 10:
                seen_titles.add(title)
                unique.append(opp)
        
        return unique

    def sort_by_urgency(self, opportunities: List[Dict]) -> List[Dict]:
        """Sort opportunities by deadline urgency"""
        def get_deadline_score(opp: Dict) -> int:
            deadline = opp.get('deadline', '').lower()
            
            if 'rolling' in deadline or 'open' in deadline:
                return 30
            
            if 'not specified' in deadline:
                return 60
            
            # Try to extract date
            date_patterns = [
                r'(\d{1,2}\s+[A-Za-z]+\s+\d{4})',
                r'([A-Za-z]+\s+\d{1,2},?\s+\d{4})',
                r'(\d{1,2}/\d{1,2}/\d{4})',
                r'(\d{4}-\d{1,2}-\d{1,2})'
            ]
            
            for pattern in date_patterns:
                match = re.search(pattern, deadline)
                if match:
                    try:
                        # Try to parse date
                        date_str = match.group(1)
                        # This is simplified - you'd need proper date parsing
                        return 15  # Assume soon if date found
                    except:
                        pass
            
            return 45
        
        return sorted(opportunities, key=get_deadline_score)
